cmake_minimum_required(VERSION 3.20)
project(uke_aloha_core VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(UKE_SANITIZE "Enable host address/undefined-behavior sanitizers" OFF)
option(UKE_SANITIZE_VPTR "Instrument C++ virtual-pointer checks when runtime supports them" ON)
add_library(uke_boot_core STATIC src/core/gpt.c src/core/policy.c src/core/dispatch.c src/core/buttons.c src/core/sha256.c src/core/fat32.c)
target_include_directories(uke_boot_core PUBLIC include)
target_compile_options(uke_boot_core PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -fno-builtin)
target_sources(uke_boot_core PRIVATE src/platform/UkePlatformPkg/PrePi/MemoryPlan.c
  src/platform/UkePlatformPkg/Sec/EntryAdmission.c src/platform/UkePlatformPkg/AndroidReturn/ReturnTransaction.c)
if(UKE_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(uke_boot_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  if(NOT UKE_SANITIZE_VPTR)
    add_compile_options(-fno-sanitize=vptr)
  endif()
endif()
add_executable(uke-esp-inspect src/host/esp_inspect.cpp)
target_link_libraries(uke-esp-inspect PRIVATE uke_boot_core)
target_compile_options(uke-esp-inspect PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
add_library(uke_boot_image STATIC src/boot/boot_image.cpp)
target_link_libraries(uke_boot_image PUBLIC uke_boot_core)
target_compile_options(uke_boot_image PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
add_executable(uke-boot-tool src/host/boot_tool.cpp)
target_link_libraries(uke-boot-tool PRIVATE uke_boot_image)
target_compile_options(uke-boot-tool PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
include(CTest)
set(UKE_LIBFDT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/upstream/dtc-7a1e017926004ecff5fce62d62d42ce9f3e00082/libfdt" CACHE PATH "Pinned libfdt source")
if(EXISTS "${UKE_LIBFDT_DIR}/libfdt.h")
  add_library(uke_libfdt STATIC)
  foreach(source fdt fdt_ro fdt_check fdt_addresses fdt_rw fdt_sw fdt_wip fdt_empty_tree)
    target_sources(uke_libfdt PRIVATE "${UKE_LIBFDT_DIR}/${source}.c")
  endforeach()
  target_include_directories(uke_libfdt SYSTEM PUBLIC "${UKE_LIBFDT_DIR}")
  add_library(uke_sec_framebuffer STATIC src/platform/UkePlatformPkg/Sec/FramebufferAdmission.c)
  target_link_libraries(uke_sec_framebuffer PUBLIC uke_libfdt)
  target_include_directories(uke_sec_framebuffer PUBLIC include)
  target_compile_options(uke_sec_framebuffer PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
  if(BUILD_TESTING)
    add_executable(uke-sec-framebuffer-tests tests/sec_framebuffer_tests.cpp)
    target_link_libraries(uke-sec-framebuffer-tests PRIVATE uke_sec_framebuffer)
    target_compile_options(uke-sec-framebuffer-tests PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
    add_test(NAME sec-framebuffer COMMAND uke-sec-framebuffer-tests)
  endif()
else()
  message(STATUS "SEC framebuffer tests unavailable: prepare pinned libfdt sources first")
endif()
if(BUILD_TESTING)
  add_executable(uke-platform-boot-tests tests/platform_boot_tests.cpp)
  target_link_libraries(uke-platform-boot-tests PRIVATE uke_boot_core)
  target_compile_options(uke-platform-boot-tests PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
  add_test(NAME platform-memory-return COMMAND uke-platform-boot-tests)
  add_executable(uke-dispatch-tests tests/dispatch_tests.cpp)
  target_link_libraries(uke-dispatch-tests PRIVATE uke_boot_core)
  target_compile_options(uke-dispatch-tests PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
  add_test(NAME dispatch COMMAND uke-dispatch-tests)
  add_executable(uke-boot-image-tests tests/boot_image_tests.cpp)
  target_link_libraries(uke-boot-image-tests PRIVATE uke_boot_image)
  target_compile_options(uke-boot-image-tests PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
  add_test(NAME boot-image COMMAND uke-boot-image-tests)
  add_test(NAME boot-tool-file-boundary COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/boot-tool-files.sh $<TARGET_FILE:uke-boot-tool>)
  add_executable(uke-fixture-disk tests/fixture_disk.cpp)
  target_compile_options(uke-fixture-disk PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
  add_executable(uke-core-tests tests/core_tests.cpp)
  target_link_libraries(uke-core-tests PRIVATE uke_boot_core)
  target_compile_options(uke-core-tests PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
  foreach(suite gpt policy config image buttons fat32 manager)
    add_test(NAME ${suite} COMMAND uke-core-tests ${suite})
  endforeach()
  add_test(NAME inspector-file-boundary COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/inspect-files.sh $<TARGET_FILE:uke-esp-inspect>)
endif()
